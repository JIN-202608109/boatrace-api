設定値 SETTING_MIN_SCORE_STAR5 = 85 SETTING_STRICT_PROB_B1 = 0.75 def clean_text(text): return str(text).strip() # 24場マッピング定義 STADIUM_IDS = { "桐生": "kiryu", "戸田": "toda", "江戸川": "edogawa", "平和島": "heiwajima", "多摩川": "tamagawa", "浜名湖": "lakehamanako", "蒲郡": "gamagori", "常滑": "tokoname", "津": "tsu", "三国": "mikuni", "びわこ": "biwako", "住之江": "suminoe", "尼崎": "amagasaki", "鳴門": "naruto", "丸亀": "marugame", "児島": "kojima", "宮島": "miyajima", "徳山": "tokuyama", "下関": "shimonoseki", "若松": "wakamatsu", "芦屋": "ashiya", "福岡": "fukuoka", "唐津": "karatsu", "大村": "omura", } # ========================================== # 🚀 Excelデータの読み込み処理（全シート＆全買い目対応） # ========================================== def load_data_from_excel(excel_path): print(f"📄 Excelファイルを読み込んでいます: {excel_path}") xls = pd.ExcelFile(excel_path) all_races_dict = {} shobu_races = {} race_results_cache = {} for sheet_name in xls.sheet_names: print(f" └ シート読み込み: {sheet_name}") df = pd.read_excel(xls, sheet_name=sheet_name) if "レースID" not in df.columns: continue grouped = df.groupby("レースID", sort=False) for race_id, group in grouped: first_row = group.iloc[0] place = clean_text(first_row["レース場"]) r_num_str = str(first_row["レース回"]).replace("R", "").strip() r_num = int(r_num_str) if r_num_str.isdigit() else 1 star = str(first_row.get("自信度", "★☆☆☆☆")) score = float(first_row.get("総合スコア", 0)) prob_b1_raw = str(first_row.get("1号艇AI勝率", "0%")).replace("%", "") try: prob_b1 = ( float(prob_b1_raw) / 100.0 if float(prob_b1_raw) > 1 else float(prob_b1_raw) ) except ValueError: prob_b1 = 0.0 shobu_flag = str(first_row.get("勝負対象", "")).strip() is_shobu = ( shobu_flag in ["〇", "○", "勝負", "TRUE", "True", "1"] ) or ( score >= SETTING_MIN_SCORE_STAR5 and prob_b1 >= SETTING_STRICT_PROB_B1 ) race_info = { "race_id": race_id, "round_num": r_num, "star": star, "score": score, "prob_b1": prob_b1, } if place not in all_races_dict: all_races_dict[place] = [] shobu_races[place] = [] if race_info not in all_races_dict[place]: all_races_dict[place].append(race_info) if is_shobu and race_info not in shobu_races[place]: shobu_races[place].append(race_info) #
